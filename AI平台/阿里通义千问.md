# 阿里通义千问 API 协议

## 1. 基础请求格式

```http
POST /api/v1/services/aigc/text-generation/generation HTTP/1.1
Host: dashscope.aliyuncs.com
Authorization: Bearer sk-xxx
Content-Type: application/json

{
  "model": "qwen-max",
  "input": {
    "messages": [
      {"role": "user", "content": "解释深度学习的原理"}
    ]
  },
  "parameters": {
    "result_format": "message"
  }
}
```

## 2. 开启思考链（Qwen-Long）

```http
POST /api/v1/services/aigc/text-generation/generation HTTP/1.1
Host: dashscope.aliyuncs.com
Authorization: Bearer sk-xxx
Content-Type: application/json
X-DashScope-SSE: enable  // 开启流式输出

{
  "model": "qwen-long",
  "input": {
    "messages": [
      {"role": "user", "content": "请详细分析：为什么需要深度学习？"}
    ]
  },
  "parameters": {
    "result_format": "message",
    "incremental_output": true,  // 增量输出
    "enable_search": true  // 启用搜索增强
  }
}
```

## 3. 流式响应报文示例

```http
HTTP/1.1 200 OK
Content-Type: text/event-stream
Cache-Control: no-cache
Connection: keep-alive

// 第一个数据块
data: {"output":{"choices":[{"message":{"role":"assistant","content":"深"},"finish_reason":null}]},"request_id":"xxx"}

// 后续数据块
data: {"output":{"choices":[{"message":{"role":"assistant","content":"度"},"finish_reason":null}]},"request_id":"xxx"}

data: {"output":{"choices":[{"message":{"role":"assistant","content":"学"},"finish_reason":null}]},"request_id":"xxx"}

// ... 更多内容

// 结束数据块
data: {"output":{"choices":[{"message":{"role":"assistant","content":"习"},"finish_reason":"stop"}]},"usage":{"total_tokens":100,"input_tokens":10,"output_tokens":90},"request_id":"xxx"}
```
